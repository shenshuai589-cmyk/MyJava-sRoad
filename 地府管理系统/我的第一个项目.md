```mysql
CREATE TABLE IF NOT EXISTS `underworld_staff` (
    `id` BIGINT NOT NULL AUTO_INCREMENT COMMENT '员工编号（主键自增）',
    `username` VARCHAR(50) NOT NULL COMMENT '登录账号',
    `password` VARCHAR(100) NOT NULL COMMENT '加密密码',
    `real_name` VARCHAR(50) NOT NULL COMMENT '职位尊称(如:勾魂使者·白无常)',
    `dept_id` INT DEFAULT NULL COMMENT '部门ID',
    `status` TINYINT DEFAULT 0 COMMENT '状态(0:在职, 1:历劫中, 2:离职)',
    PRIMARY KEY (`id`),
    UNIQUE KEY `uk_username` (`username`) COMMENT '账号唯一索引'
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_general_ci COMMENT='阴差员工表';
```

**4.统一企业级响应结果与异常流转**
```java
package com.powernode.underworld.common;

import lombok.Data;

@Data
public class Result<T> {
    private Integer code; // 200 成功，500 失败
    private String msg;   // 提示信息
    private T data;       // 返回的具体数据（比如单个阴差、或者阴差列表）

    public static <T> Result<T> success(T data) {
        Result<T> result = new Result<>();
        result.setCode(200);
        result.setMsg("地府政务流转成功");
        result.setData(data);
        return result;
    }

    public static <T> Result<T> success() {
        return success(null);
    }

    public static <T> Result<T> error(String msg) {
        Result<T> result = new Result<>();
        result.setCode(500);
        result.setMsg(msg);
        return result;
    }
}
```

**5.编写mapper层

> 在com.powernode.underworlds路径下新建一个包为mapper

UnderworldStaffMapper
```java
package com.powernode.underworlds.mapper;  
  
import com.powernode.underworlds.pojo.UnderworldStaff;  
import org.apache.ibatis.annotations.Mapper;  
import org.apache.ibatis.annotations.Param;  
import java.util.List;  
  
/**  
 * 阴差数据访问层接口  
 */  
@Mapper // 声明这是一个MyBatis的Mapper接口  
public interface UnderworldStaffMapper {  
  
    /**  
     * C (Create): 录入新阴差  
     * @param staff 阴差信息  
     * @return 影响的行数（大于0表示成功）  
     */  
    int insertStaff(UnderworldStaff staff);  
  
    /**  
     * R (Read): 根据工号(ID)查询单个阴差  
     * @param id 员工编号  
     * @return 阴差对象  
     */  
    UnderworldStaff selectById(Long id);  
  
    /**  
     * R (Read): 动态条件查询阴差列表（支持按部门或状态筛选）  
     * @param deptId 部门ID  
     * @param status 状态  
     * @return 阴差列表  
     */  
    List<UnderworldStaff> selectList(@Param("deptId") Integer deptId, @Param("status") Byte status);  
  
    /**  
     * U (Update): 动态修改阴差资料  
     * @param staff 包含要修改的信息（必须带上id）  
     * @return 影响的行数  
     */  
    int updateStaff(UnderworldStaff staff);  
  
    /**  
     * D (Delete): 根据工号物理删除  
     * @param id 员工编号  
     * @return 影响的行数  
     */  
    int deleteById(Long id);  
}
```
==注意：在这个接口上，写的一般是数据库CRUD的方法，当方法的参数有多个时，可以使用@Param注解进行标注==

```xml
            <if test="realName != null">real_name = #{realName},</if>  
            <if test="deptId != null">dept_id = #{deptId},</if>
```

**8.编写service接口的实现类**
UnderworldStaffServiceImpl.java
```java
package com.powernode.underworlds.service.impl;  
  
import com.powernode.underworlds.mapper.UnderworldStaffMapper;  
import com.powernode.underworlds.pojo.UnderworldStaff;  
import com.powernode.underworlds.service.UnderworldStaffService;  
import org.springframework.beans.factory.annotation.Autowired;  
import org.springframework.stereotype.Service;  
import java.util.List;  
  
/**  
 * 阴差业务逻辑层实现类  
 */  
@Service // 极其重要：告诉Spring这是一个Service组件，自动注入到Spring容器中  
public class UnderworldStaffServiceImpl implements UnderworldStaffService {  
  
    @Autowired // 自动注入我们刚才写好的 Mapper 接口  
    private UnderworldStaffMapper staffMapper;  
  
    @Override  
    public boolean saveStaff(UnderworldStaff staff) {  
        // 企业级业务扩展点：后续可以在这里给密码进行 MD5/BCrypt 加密  
        // Mapper返回的是受影响行数，大于0表示成功插入  
        return staffMapper.insertStaff(staff) > 0;  
    }  
  
    @Override  
    public UnderworldStaff getStaffById(Long id) {  
        return staffMapper.selectById(id);  
    }  
  
    @Override  
    public List<UnderworldStaff> getStaffList(Integer deptId, Byte status) {  
        return staffMapper.selectList(deptId, status);  
    }  
  
    @Override  
    public boolean renewStaff(UnderworldStaff staff) {  
        return staffMapper.updateStaff(staff) > 0;  
    }  
  
    @Override  
    public boolean removeStaff(Long id) {  
        return staffMapper.deleteById(id) > 0;  
    }  
}
```

**9.编写controller层**
> 在src/main/java/com/powernode/underworld新建一个包叫controller

在该包下新建一个UnderworldStaffController类
UnderworldStaffController.java
```java
package com.powernode.underworlds.controller;  
  
import com.powernode.underworlds.common.Result;  
import com.powernode.underworlds.pojo.UnderworldStaff;  
import com.powernode.underworlds.service.UnderworldStaffService;  
import org.springframework.beans.factory.annotation.Autowired;  
import org.springframework.web.bind.annotation.*;  
import java.util.List;  
  
/**  
 * 阴差管理控制层  
 */  
@RestController // 1. 声明这是一个REST风格的Controller，所有方法返回值都会自动转成 JSON 格式  
@RequestMapping("/staff") // 2. 抽取公共请求路径，所有阴差相关的接口都以 /staff 开头  
public class UnderworldStaffController {  
  
    @Autowired // 自动注入业务层大脑  
    private UnderworldStaffService staffService;  
  
    /**  
     * C (Create): 录入新阴差  
     * 路径：POST http://localhost:8080/staff  
     * @RequestBody: 表示接收前端传过来的 JSON 字符串，并自动转换成 Java 对象  
     */  
    @PostMapping  
    public Result<String> addStaff(@RequestBody UnderworldStaff staff) {  
        boolean success = staffService.saveStaff(staff);  
        return success ? Result.success("新阴差成功录入生死簿！") : Result.error("录入失败，请核对信息");  
    }  
  
    /**  
     * R (Read): 根据ID查询单个阴差  
     * 路径：GET http://localhost:8080/staff/1  
     * @PathVariable: 用于接收路径上的参数（把 URL 里的 {id} 传给方法的 Long id）  
     */  
    @GetMapping("/{id}")  
    public Result<UnderworldStaff> getStaff(@PathVariable Long id) {  
        UnderworldStaff staff = staffService.getStaffById(id);  
        if (staff == null) {  
            return Result.error("查无此人，该工号未曾登记在册");  
        }  
        return Result.success(staff);  
    }  
  
    /**  
     * R (Read): 条件查询阴差列表  
     * 路径：GET http://localhost:8080/staff/list?deptId=3  
     * @RequestParam: 接收传统的问号传参，required = false 表示前端如果不传这个参数也不会报错  
     */  
    @GetMapping("/list")  
    public Result<List<UnderworldStaff>> listStaff(@RequestParam(required = false) Integer deptId,  
                                                   @RequestParam(required = false) Byte status) {  
        List<UnderworldStaff> list = staffService.getStaffList(deptId, status);  
        return Result.success(list);  
    }  
  
    /**  
     * U (Update): 修改阴差资料  
     * 路径：PUT http://localhost:8080/staff  
     */    @PutMapping  
    public Result<String> updateStaff(@RequestBody UnderworldStaff staff) {  
        boolean success = staffService.renewStaff(staff);  
        return success ? Result.success("阴差资料生死簿同步成功！") : Result.error("修改失败，查无此人");  
    }  
  
    /**  
     * D (Delete): 根据工号彻底除名  
     * 路径：DELETE http://localhost:8080/staff/1  
     */    @DeleteMapping("/{id}")  
    public Result<String> deleteStaff(@PathVariable Long id) {  
        boolean success = staffService.removeStaff(id);  
        return success ? Result.success("该阴差已从地府彻底除名，打入凡间！") : Result.error("除名失败，工号不存在");  
    }  
}
```
