
### 1. \__str__（toString）

```python
class Student:  
  
    name  = None  
    age = None  
    gender = None  
  
    def __init__(self,name,age,gender):  
        self.name = name  
        self.age = age  
        self.gender = gender  
  
  
stu = Student("张三",23,"男")  
print(stu) # 我们在打印实例对象时，输出的是地址值<__main__.Student object at 0x000002335644D7F0>
```

```python
class Student:  
  
    name = None  
    age = None  
  
    def __init__(self,name,age):  
        self.name = name  
        self.age = age  
  
    def __str__(self):  
        return f"Student类对象,name={self.name},age={self.age}"  
  
  
  
  
# __str__  
  
stu1 = Student("张三",23)  
print(stu1)
```

### 2. /__lt__小于符号比较方法

```python
class Student:  
  
    name = None  
    age = None  
  
    def __init__(self,name,age):  
        self.name = name  
        self.age = age  
  
    # __lt__  
    def __lt__(self,other):  
        return self.age < other.age  
  
  
stu1 = Student("张三",23)  
stu2 = Student("李四",24)  
print(stu1<stu2)
```



### 3. \__le__ (小于等于、大于等于)

```python
class Student:  
  
    name = None  
    age = None  
  
    def __init__(self,name,age):  
        self.name = name  
        self.age = age  
   

    # __le__  
    def __le__(self,other):  
        return self.age <= other.age  
  
stu1 = Student("张三",23)  
stu3 = Student("王三",23)  
print(stu1>=stu3)
```
